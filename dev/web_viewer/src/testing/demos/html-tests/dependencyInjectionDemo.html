Example: How Dependency Injection Works

// 1. ResourceManager provides shared contexts
const resourceManager = new ResourceManager();
await resourceManager.initialize();

// 2. Modules receive injected dependencies
const whisperModule = new WhisperModule({
    audioContext: resourceManager.audioContext,
    transformersEnv: resourceManager.transformersEnv,
    resourceManager: resourceManager
});

// 3. Models use shared resources
const pipeline = await whisperModule.getPipeline('automatic-speech-recognition', 'Xenova/whisper-tiny.en');
